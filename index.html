🤖 Greeting Robot

Program By Abhiraj

Project Behaviour

👤 Person aata hai → Sensor detect karta hai
🎵 Welcome audio play hota hai
🙏 Hands Namaste position mein aate hain
🔔 Buzzer beep karta hai

🚶 Person chala jata hai → Goodbye audio play hota hai
🤖 Hands neeche chale jate hain
🔄 Robot next person ke liye ready ho jata hai

Arduino Code

#include <SoftwareSerial.h>
          #include <DFRobotDFPlayerMini.h>
          #include <Servo.h>

          const int TRIG_PIN = 7;
          const int ECHO_PIN = 8;

          const int LEFT_SERVO_PIN = 5;
          const int RIGHT_SERVO_PIN = 6;

          const int BUZZER_PIN = 4;

          const int DFPLAYER_RX_PIN = 10;
          const int DFPLAYER_TX_PIN = 11;

          const int DETECTION_DISTANCE = 150;

          const unsigned long DETECT_CONFIRM_TIME = 700;
          const unsigned long LEAVE_CONFIRM_TIME = 2000;

          const int LEFT_HAND_DOWN = 20;
          const int RIGHT_HAND_DOWN = 160;

          const int LEFT_HAND_JOINED = 110;
          const int RIGHT_HAND_JOINED = 70;

          Servo leftServo;
          Servo rightServo;

          SoftwareSerial dfSerial(
          DFPLAYER_RX_PIN,
          DFPLAYER_TX_PIN
          );

          DFRobotDFPlayerMini dfPlayer;

          bool personPresent = false;

          unsigned long detectedStartTime = 0;
          unsigned long absentStartTime = 0;

          bool detectionTimerRunning = false;
          bool absenceTimerRunning = false;

          void setup() {

          Serial.begin(9600);

          pinMode(TRIG_PIN, OUTPUT);
          pinMode(ECHO_PIN, INPUT);

          pinMode(BUZZER_PIN, OUTPUT);
          digitalWrite(BUZZER_PIN, LOW);

          leftServo.attach(LEFT_SERVO_PIN);
          rightServo.attach(RIGHT_SERVO_PIN);

          moveHandsDown();

          dfSerial.begin(9600);

          if (dfPlayer.begin(dfSerial)) {
          dfPlayer.volume(25);
          }

          delay(1000);
          }

          void loop() {

          int distance = getDistance();

          bool objectDetected =
          (distance > 0 &&
          distance <= DETECTION_DISTANCE);

          if (objectDetected) {

          absenceTimerRunning = false;

          if (!personPresent) {

          if (!detectionTimerRunning) {

          detectedStartTime = millis();
          detectionTimerRunning = true;
          }

          if (millis() - detectedStartTime
          >= DETECT_CONFIRM_TIME) {

          personPresent = true;
          detectionTimerRunning = false;

          welcomePerson();
          }
          }
          }

          else {

          detectionTimerRunning = false;

          if (personPresent) {

          if (!absenceTimerRunning) {

          absentStartTime = millis();
          absenceTimerRunning = true;
          }

          if (millis() - absentStartTime
          >= LEAVE_CONFIRM_TIME) {

          personPresent = false;
          absenceTimerRunning = false;

          goodbyePerson();
          }
          }
          }

          delay(50);
          }
          int getDistance() {

          digitalWrite(TRIG_PIN, LOW);
          delayMicroseconds(2);

          digitalWrite(TRIG_PIN, HIGH);
          delayMicroseconds(10);

          digitalWrite(TRIG_PIN, LOW);

          long duration =
          pulseIn(ECHO_PIN, HIGH, 30000);

          if (duration == 0) {
          return -1;
          }

          int distance =
          duration * 0.034 / 2;

          return distance;
          }

          void welcomePerson() {

          dfPlayer.playMp3Folder(1);

          moveHandsJoined();

          beepGreeting();
          }

          void goodbyePerson() {

          dfPlayer.playMp3Folder(2);

          moveHandsDown();

          digitalWrite(BUZZER_PIN, LOW);
          }

          void moveHandsJoined() {

          for (int i = 0; i <= 90; i++) { int leftPosition=map( i, 0, 90, LEFT_HAND_DOWN, LEFT_HAND_JOINED ); int
            rightPosition=map( i, 0, 90, RIGHT_HAND_DOWN, RIGHT_HAND_JOINED ); leftServo.write(leftPosition);
            rightServo.write(rightPosition); delay(15); } } void moveHandsDown() { for (int i=0; i <=90; i++) { int
            leftPosition=map( i, 0, 90, LEFT_HAND_JOINED, LEFT_HAND_DOWN ); int rightPosition=map( i, 0, 90,
            RIGHT_HAND_JOINED, RIGHT_HAND_DOWN ); leftServo.write(leftPosition); rightServo.write(rightPosition);
            delay(15); } } void beepGreeting() { for (int i=0; i < 3; i++) { digitalWrite(BUZZER_PIN, HIGH); delay(120);
            digitalWrite(BUZZER_PIN, LOW); delay(120); } }
